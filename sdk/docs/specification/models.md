```python
class ContentFormat(StrEnum):
    """The shape of a Document's content string, stamped in metadata.

    Every parser stamps metadata["content_format"] with one of these
    members so a downstream step can dispatch on what kind of string
    content actually is, rather than guessing from metadata["content_type"]
    (the source's media type, not what the parser turned it into). Being a
    StrEnum, every member serializes to a plain string wherever metadata is
    persisted (a cache's JSONB column, Milvus's JSON metadata field).
    """

    TEXT = "text"
    MARKDOWN = "markdown"
    DOCLING = "docling"


@dataclass(frozen=True, slots=True)
class Source:
    """
    Represents a raw piece of externally sourced content.
    A Source is produced by a Fetcher and contains the original bytes together with enough metadata to identify, validate, cache, and later transform the source into one or more Documents.
    Sources are independently cacheable and their freshness may be validated
    against the originating system using ETags, modification timestamps,
    content hashes, or equivalent mechanisms.
    """

    id: str
    uri: str
    content: bytes
    content_type: str
    content_hash: str
    etag: str | None = None
    last_modified: datetime | None = None
    metadata: dict[str, Any]
    last_validated: datetime | None = None  # Persisted; round-tripped by the cache

    cached: bool = False  # Not in the database


@dataclass(frozen=True, slots=True)
class Document:
    """Represents a text-based document or chunk processed along the pipeline.

    Attributes:
        id: Unique identifier for this unit (typically a content/configuration hash).
        source_id: Normalized URI of the originating Source.
        content: Text content of the document or chunk.
        step_config_hash: Hash representing the upstream transform configuration.
        metadata: Extended metadata dictionary tracking lineage, chunk offsets,
            URL filters, and parser outputs. Reserved keys: "source_uri",
            "content_type" (the originating Source's URI and media type),
            and "content_format" (a ContentFormat value describing what kind
            of string content is) -- all three stamped by
            Parser.build_document so no parser can forget them.
        embedding: Dense vector representation generated by Embedder steps.
        cached: Whether this instance was served from the cache rather than
            freshly produced. Runtime-only: never persisted.
    """

    id: str
    source_id: str
    content: str
    step_config_hash: str = ""
    embedding: list[float] | None = None
    metadata: dict[str, Any] = field(default_factory=dict)

    cached: bool = False  # Not in the database
```
